function Format-DFIRDate($d){
    if ($null -eq $d -or $d -eq "" -or $d -eq "N/A") { return "N/A" }
    try {
        $dtObj = if ($d -is [datetime]) { $d } else { [datetime]::Parse($d.ToString()) }
        if ($dtObj.Year -lt 2000 -or $dtObj.Year -gt 2035) { return "N/A" }
        return $dtObj.ToString("yyyy-MM-dd HH:mm:ss")
    } catch { return "N/A" }
}

function G-E($n){ 
    G-H "Events_$n" { 
        # Optional: Increase -Newest 1000 to 5000 if you need to search further back
        $ev = Get-EventLog -LogName $n -Newest 5000 -EA 0
        $res = $ev | % {
            $tGen = Format-DFIRDate $_.TimeGenerated
            
            # 1. Grab primary UserName property (populated in Application/System logs)
            $uName = $_.UserName

            # 2. Fallback: Parse Account/User Name from Security Log message text if blank
            if ([string]::IsNullOrWhiteSpace($uName) -and $_.Message) {
                if ($_.Message -match "(?i)(?:Account Name|User Name|SubjectUserName):\s*([^\s\r\n]+)") {
                    $candidate = $matches[1].Trim()
                    # Ignore null/anonymous indicators like '-'
                    if ($candidate -and $candidate -ne "-") {
                        $uName = $candidate
                    }
                }
            }

            # 3. Default to N/A if no user context was logged
            if ([string]::IsNullOrWhiteSpace($uName)) { $uName = "N/A" }

            # Push to Supertimeline CSV
            Add-TL -t $tGen -macb "M..." -a "EventLog-$n" -u $uName -d "ID: $($_.EventID) | $($_.Message)" -mod $tGen
            
            # Push to HTML Report Table
            [pscustomobject]@{ TimeGenerated=$tGen; EventID=$_.EventID; User=$uName; Message=$_.Message }
        }
        $res | ConvertTo-Html -Fragment 
    }

G-H "Prefetch" { 
    $pfState = (Get-ItemProperty "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager\Memory Management\PrefetchParameters" -Name EnablePrefetcher -EA 0).EnablePrefetcher
    
    $headerMsg = ""
    if ($null -eq $pfState -or $pfState -eq 0) {
        $headerMsg = "<p style='color:#b30000;'><b>Note:</b> Windows Prefetch is disabled on this machine (EnablePrefetcher = 0). This is default behavior for Windows Server.</p>"
    }

    $pf = Get-ChildItem C:\Windows\Prefetch\*.pf -ErrorAction SilentlyContinue | Sort-Object LastWriteTime -Descending
    
    if (-not $pf) {
        return "$headerMsg<p>No .pf files found in C:\Windows\Prefetch\</p>"
    }

    $sha256 = [System.Security.Cryptography.HashAlgorithm]::Create('SHA256')

    $res = $pf | ForEach-Object { 
        $created  = Format-DFIRDate $_.CreationTime
        $modified = Format-DFIRDate $_.LastWriteTime
        $accessed = Format-DFIRDate $_.LastAccessTime
        $sizeKB   = [math]::Round($_.Length / 1KB, 2)

        # Compute SHA256 Hash
        $hashBytes = $sha256.ComputeHash([System.IO.File]::ReadAllBytes($_.FullName))
        $hash = ([System.BitConverter]::ToString($hashBytes)).Replace('-', '').ToLower()
        
        # Add to DFIR SuperTimeline
        if($modified -ne "N/A"){ Add-TL -t $modified -macb "M..." -a "Prefetch-Modified" -u "SYSTEM" -d "$($_.Name) (Hash: $hash)" -mod $modified -birth $created -acc $accessed }
        if($created -ne "N/A"){  Add-TL -t $created  -macb "...B" -a "Prefetch-Created"  -u "SYSTEM" -d "$($_.Name) (Hash: $hash)" -birth $created -mod $modified -acc $accessed }
        
        [pscustomobject]@{ 
            PrefetchFile = $_.Name
            SizeBytesKB  = $sizeKB
            CreatedTime  = $created
            ModifiedTime = $modified
            AccessedTime = $accessed
            SHA256       = $hash
        }
    }
    "$headerMsg" + ($res | ConvertTo-Html -Fragment)
}
