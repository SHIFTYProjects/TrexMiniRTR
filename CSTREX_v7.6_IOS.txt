#!/usr/bin/zsh
# TREX macOS Forensic Incident Response Collector (Full DFIR Version)
# Optimized for EDR execution (CrowdStrike RTR / SentinelOne Remote Shell) with up to 2-minute timeouts.

if [ "$EUID" -ne 0 ]; then
    echo "[-] Execution halted: Root privileges required."
    exit 1
fi

HOST=$(hostname)
DATE_STR=$(date "+%Y-%m-%d %H:%M:%S")
REPORT_DIR="/tmp/${HOST}_Trex_Reports"
mkdir -p "$REPORT_DIR"
TIMELINE_CSV="$REPORT_DIR/SuperTimeline.csv"

# Initialize SuperTimeline
echo '"Timestamp","MACB","Modified","Access","Change","Birth","Artifact","User","Details","Last Run","Next Run"' > "$TIMELINE_CSV"

add_tl() {
    local t="$1" macb="$2" art="$3" user="$4" det="$5" mod="${6:-N/A}" acc="${7:-N/A}" chg="${8:-N/A}" birth="${9:-N/A}"
    [ -z "$t" ] || [ "$t" = "N/A" ] && return
    local clean_det=$(echo "$det" | tr '\r\n' ' ' | sed 's/"/""/g')
    echo "\"$t\",\"$macb\",\"$mod\",\"$acc\",\"$chg\",\"$birth\",\"$art\",\"$user\",\"$clean_det\",\"N/A\",\"N/A\"" >> "$TIMELINE_CSV"
}

generate_html_report() {
    local title="$1"
    local content="$2"
    local outfile="$REPORT_DIR/${title}.html"
    cat <<EOF > "$outfile"
<html>
<head>
<style>
body{margin:0;font-family:sans-serif;background:#eee}
.hb{background:linear-gradient(135deg,#1e3c72 0%,#2a5298 100%);color:#fff;padding:20px;border-bottom:4px solid #b30000;margin-bottom:15px}
.hb h2{margin:0 0 5px 0}
.c{padding:0 15px 15px 15px}
table{width:100%;border-collapse:collapse;background:#fff}
th,td{border:1px solid #ccc;padding:8px;font-size:12px;text-align:left}
th{background:#1e3c72;color:#fff}
pre{background:#fff;border:1px solid #ccc;padding:10px;overflow-x:auto;font-size:11px}
</style>
</head>
<body>
<div class='hb'><h2>TREX REPORT: ${title}</h2>Target: ${HOST}</div>
<div class='c'>
${content}
<br><small style='color:#555;font-weight:bold;'>Context: REAL TIME RESPONSE - macOS DATA PULL - $DATE_STR</small>
</div>
</body>
</html>
EOF
}

# --- 1. PCInfo ---
SYS_MODEL=$(sysctl -n hw.model)
CPU_NAME=$(sysctl -n machdep.cpu.brand_string 2>/dev/null || echo "Apple Silicon / Generic CPU")
RAM_BYTES=$(sysctl -n hw.memsize)
RAM_GB=$((RAM_BYTES / 1024 / 1024 / 1024))
OS_VER=$(sw_vers -productVersion)
BUILD_VER=$(sw_vers -buildVersion)
BOOT_TIME=$(sysctl -n kern.boottime | awk '{print $4}' | tr -d ',')
BOOT_DATE=$(date -r "$BOOT_TIME" "+%Y-%m-%d %H:%M:%S" 2>/dev/null || echo "N/A")

PC_INFO_HTML="<table>
<tr><th>Host</th><td>${HOST}</td></tr>
<tr><th>Domain / Network</th><td>$(domainname 2>/dev/null || echo "N/A")</td></tr>
<tr><th>OS</th><td>macOS Version ${OS_VER}</td></tr>
<tr><th>Version / Build</th><td>Build ${BUILD_VER}</td></tr>
<tr><th>Manufacturer / Model</th><td>Apple Inc. - ${SYS_MODEL}</td></tr>
<tr><th>CPU</th><td>${CPU_NAME}</td></tr>
<tr><th>Total RAM</th><td>${RAM_GB} GB</td></tr>
<tr><th>Last Boot Time</th><td>${BOOT_DATE}</td></tr>
</table>"
generate_html_report "PCInfo" "$PC_INFO_HTML"
add_tl "$BOOT_DATE" "M..." "SystemBoot" "SYSTEM" "System Last Boot Time" "$BOOT_DATE"

# --- 2. Services ---
generate_html_report "Services" "<pre>$(launchctl list)</pre>"

# --- 3. Processes ---
generate_html_report "Processes" "<pre>$(ps aux -r)</pre>"

# --- 4. Software ---
APPS_LIST=$(ls -laT /Applications /System/Applications /Users/*/Applications 2>/dev/null)
generate_html_report "Software" "<pre>${APPS_LIST}</pre>"

# --- 5. Netstat ---
generate_html_report "Netstat" "<pre>$(netstat -anv 2>/dev/null)</pre>"

# --- 6. IPConfig ---
generate_html_report "IPConfig" "<pre>$(ifconfig -a)</pre>"

# --- 7. Disk ---
generate_html_report "Disk" "<pre>$(df -h; echo "\n=== Diskutil List ==="; diskutil list)</pre>"

# --- 8. Drivers (Kernel & System Extensions) ---
generate_html_report "Drivers" "<pre>=== System Extensions ===\n$(systemextensionsctl list 2>/dev/null)\n\n=== Kernel Extensions ===\n$(kmutil showloaded 2>/dev/null || kextstat 2>/dev/null)</pre>"

# --- 9. Startup & Persistence ---
PERSIST_DATA="=== LaunchDaemons ===\n$(ls -laT /Library/LaunchDaemons /System/Library/LaunchDaemons 2>/dev/null)\n\n=== LaunchAgents ===\n$(ls -laT /Library/LaunchAgents /System/Library/LaunchAgents /Users/*/Library/LaunchAgents 2>/dev/null)\n\n=== Crontabs ===\n$(crontab -l 2>/dev/null || echo 'No root cron')"
generate_html_report "Startup" "<pre>${PERSIST_DATA}</pre>"

# --- 10. Tasks (Cron & Launchd Jobs) ---
generate_html_report "Tasks" "<pre>${PERSIST_DATA}</pre>"

# --- 11. Shares ---
generate_html_report "Shares" "<pre>$(smbutil statshares -a 2>/dev/null || sharing -l 2>/dev/null || echo 'No active SMB shares')</pre>"

# --- 12. Updates (Restored Full Software Update History) ---
UPDATE_HIST=$(softwareupdate --history 2>/dev/null || pkgutil --pkgs)
generate_html_report "Updates" "<pre>${UPDATE_HIST}</pre>"

# --- 13. EnvVars ---
generate_html_report "EnvVars" "<pre>$(printenv)</pre>"

# --- 14. USB (Restored Full Hardware Profiling) ---
USB_DATA=$(system_profiler SPUSBDataType 2>/dev/null || ioreg -p IOUSB -l -w 0)
generate_html_report "USB" "<pre>${USB_DATA}</pre>"

# --- 15. ARP ---
generate_html_report "ARP" "<pre>$(arp -a)</pre>"

# --- 16. Route ---
generate_html_report "Route" "<pre>$(netstat -rn)</pre>"

# --- 17. DNS ---
generate_html_report "DNS" "<pre>$(scutil --dns)</pre>"

# --- 18. Security Posture (FileVault, Gatekeeper, SIP, TCC) ---
FV_STATUS=$(fdesetup status 2>/dev/null)
SIP_STATUS=$(csrutil status 2>/dev/null)
GK_STATUS=$(spctl --status 2>/dev/null)
generate_html_report "Bitlocker" "<pre>FileVault: ${FV_STATUS}\nSIP: ${SIP_STATUS}\nGatekeeper: ${GK_STATUS}</pre>"

# --- 19. Power ---
generate_html_report "Power" "<pre>$(pmset -g live 2>/dev/null; echo "\n=== Assertions ==="; pmset -g assertions 2>/dev/null)</pre>"

# --- 20. Firewall ---
FW_STATUS=$(socketfilterfw --getglobalstate 2>/dev/null || echo "Firewall query unavailable")
generate_html_report "Firewall" "<pre>${FW_STATUS}\n\n=== Rules ===\n$(socketfilterfw --listapps 2>/dev/null)</pre>"

# --- 21. Network Adapters ---
generate_html_report "Network" "<pre>$(networksetup -listallhardwareports 2>/dev/null)</pre>"

# --- 22. Uptime ---
generate_html_report "Uptime" "<pre>$(uptime)</pre>"

# --- 23. Reliability / Unified Logs (Restored 24-Hour Security/Error Window) ---
LOG_SAMPLES=$(log show --predicate 'eventMessage contains "error" OR eventMessage contains "fail" OR process == "sudo" OR process == "tccd"' --last 24h 2>/dev/null | tail -n 500 || echo "Unified Log query skipped")
generate_html_report "Reliability" "<pre>${LOG_SAMPLES}</pre>"

# --- 24. HostsFile ---
generate_html_report "HostsFile" "<pre>$(cat /etc/hosts)</pre>"

# --- 25. RunKeys / Background Task Management (BTM) ---
BTM_DATA=$(plutil -p /var/db/com.apple.backgroundtaskmanagement/ServiceManagement.plist 2>/dev/null || plutil -p /Users/*/Library/Preferences/com.apple.loginitems.plist 2>/dev/null || echo "N/A")
generate_html_report "RunKeys" "<pre>${BTM_DATA}</pre>"

# --- 26. Shell Histories (Zsh, Bash, Root) ---
HIST_LOG=""
for user_dir in /Users/* /var/root; do
    u=$(basename "$user_dir")
    for hfile in "$user_dir/.zsh_history" "$user_dir/.bash_history" "$user_dir/.sh_history"; do
        if [ -f "$hfile" ]; then
            mtime=$(date -r "$hfile" "+%Y-%m-%d %H:%M:%S" 2>/dev/null)
            add_tl "$mtime" "M..." "ShellHistory" "$u" "History file modified: $hfile" "$mtime"
            HIST_LOG+="=== User: $u ($hfile) ===\n$(cat "$hfile" 2>/dev/null)\n\n"
        fi
    done
done
generate_html_report "PSConsoleHist" "<pre>${HIST_LOG}</pre>"

# --- 27. TypedPaths (Shared File Lists / Recents) ---
RECENT_DATA=$(ls -laT /Users/*/Library/Application\ Support/com.apple.sharedfilelist/* 2>/dev/null || echo "N/A")
generate_html_report "TypedPaths" "<pre>${RECENT_DATA}</pre>"

# --- 28. WifiPass ---
WIFI_PROFILES=$(networksetup -listpreferredwirelessnetworkhardwareport en0 2>/dev/null || echo "N/A")
generate_html_report "WifiPass" "<pre>${WIFI_PROFILES}</pre>"

# --- 29. WMIPersistence (System Extensions & Daemon Bindings) ---
SYS_EXTS=$(systemextensionsctl list 2>/dev/null || echo "N/A")
generate_html_report "WMIPersistence" "<pre>${SYS_EXTS}</pre>"

# --- 30. Downloads & LSQuarantine (Equivalent to Windows Zone Identifier) ---
DL_REPORT=""
QUARANTINE_DIR="$REPORT_DIR/QuarantineDBs"
mkdir -p "$QUARANTINE_DIR"

for user_dir in /Users/*; do
    u=$(basename "$user_dir")
    
    # Download directory enumeration
    if [ -d "$user_dir/Downloads" ]; then
        for item in "$user_dir/Downloads"/*; do
            if [ -e "$item" ]; then
                ctime=$(date -r "$item" "+%Y-%m-%d %H:%M:%S" 2>/dev/null)
                add_tl "$ctime" "...B" "Download-Created" "$u" "$(basename "$item")" "" "" "" "$ctime"
                DL_REPORT+="$u | $(basename "$item") | $ctime\n"
            fi
        done
    fi
    
    # LSQuarantine database extraction
    qdb="$user_dir/Library/Preferences/com.apple.LaunchServices.QuarantineEventsV2"
    if [ -f "$qdb" ]; then
        cp "$qdb" "$QUARANTINE_DIR/${u}_QuarantineEventsV2.db" 2>/dev/null
    fi
done
generate_html_report "Downloads" "<pre>${DL_REPORT}</pre>"

# --- 31. Profiles ---
USERS_LIST=$(dscl . -list /Users | grep -v '^_'; echo "\n=== User Folder Creation Dates ==="; ls -laT /Users)
generate_html_report "Profiles" "<pre>${USERS_LIST}</pre>"

# --- 32. Web Extensions & Browser Artifact Collection ---
WEB_HIST_DIR="$REPORT_DIR/WebHist"
mkdir -p "$WEB_HIST_DIR"

for user_dir in /Users/*; do
    u=$(basename "$user_dir")
    
    # Chrome
    c_path="$user_dir/Library/Application Support/Google/Chrome"
    if [ -d "$c_path" ]; then
        find "$c_path" -name "History" -exec cp {} "$WEB_HIST_DIR/${u}_Chrome_History.db" \; 2>/dev/null
        find "$c_path" -name "Cookies" -exec cp {} "$WEB_HIST_DIR/${u}_Chrome_Cookies.db" \; 2>/dev/null
        find "$c_path" -name "Login Data" -exec cp {} "$WEB_HIST_DIR/${u}_Chrome_LoginData.db" \; 2>/dev/null
        find "$c_path" -name "Shortcuts" -exec cp {} "$WEB_HIST_DIR/${u}_Chrome_Shortcuts.db" \; 2>/dev/null
        find "$c_path" -name "Bookmarks" -exec cp {} "$WEB_HIST_DIR/${u}_Chrome_Bookmarks" \; 2>/dev/null
    fi
    
    # Safari
    s_path="$user_dir/Library/Safari"
    if [ -d "$s_path" ]; then
        cp "$s_path/History.db" "$WEB_HIST_DIR/${u}_Safari_History.db" 2>/dev/null
        cp "$s_path/Bookmarks.plist" "$WEB_HIST_DIR/${u}_Safari_Bookmarks.plist" 2>/dev/null
        cp "$s_path/Downloads.plist" "$WEB_HIST_DIR/${u}_Safari_Downloads.plist" 2>/dev/null
    fi
    
    # Firefox
    f_path="$user_dir/Library/Application Support/Firefox/Profiles"
    if [ -d "$f_path" ]; then
        find "$f_path" -name "places.sqlite" -exec cp {} "$WEB_HIST_DIR/${u}_Firefox_places.sqlite" \; 2>/dev/null
        find "$f_path" -name "cookies.sqlite" -exec cp {} "$WEB_HIST_DIR/${u}_Firefox_cookies.sqlite" \; 2>/dev/null
        find "$f_path" -name "logins.json" -exec cp {} "$WEB_HIST_DIR/${u}_Firefox_logins.json" \; 2>/dev/null
    fi
done
generate_html_report "WebExt" "<p>Extracted raw SQLite databases to WebHist/ directory.</p>"

# --- 33. Log Archive & Installation Logs Collection ---
LOG_DIR="$REPORT_DIR/EvtLogs"
mkdir -p "$LOG_DIR"
cp /var/log/system.log* "$LOG_DIR/" 2>/dev/null
cp /var/log/install.log* "$LOG_DIR/" 2>/dev/null
cp /var/log/wifi.log* "$LOG_DIR/" 2>/dev/null

# Extract raw BSM Audit logs
if [ -d "/var/audit" ]; then
    mkdir -p "$LOG_DIR/audit"
    cp /var/audit/* "$LOG_DIR/audit/" 2>/dev/null
fi

# --- 34. Archive Packaging ---
ZIP_PATH="/tmp/${HOST}_TREX.zip"
rm -f "$ZIP_PATH"
zip -r "$ZIP_PATH" "$REPORT_DIR" >/dev/null 2>&1
rm -rf "$REPORT_DIR"

echo "[+] Forensic collection completed: ${ZIP_PATH}"
